import { apiClient, AppApiError } from '@/lib/api-client';
import type {
  Student,
  StudentInput,
  StudentsListResponse,
  ListStudentsParams,
} from '@/features/students/types';

export async function listStudents(params: ListStudentsParams = {}) {
  const searchParams = new URLSearchParams();
  if (params.page !== undefined) searchParams.set('page', String(params.page));
  if (params.limit !== undefined) searchParams.set('limit', String(params.limit));
  if (params.search) searchParams.set('search', params.search);
  if (params.sort) searchParams.set('sort', params.sort);
  if (params.grade) searchParams.set('grade', params.grade);
  if (params.class) searchParams.set('class', params.class);
  if (params.status) searchParams.set('status', params.status);
  if (params.dietary) searchParams.set('dietary', params.dietary);
  if (params.mealEligibility)
    searchParams.set('mealEligibility', params.mealEligibility);

  const qs = searchParams.toString();
  return apiClient<StudentsListResponse>(qs ? `/students?${qs}` : '/students');
}

export async function getStudentById(id: string) {
  return apiClient<Student>(`/students/${id}`);
}

export async function createStudent(payload: StudentInput) {
  return apiClient<Student>('/students', {
    method: 'POST',
    body: JSON.stringify(payload),
  });
}

export async function updateStudent(id: string, payload: Partial<StudentInput>) {
  return apiClient<Student>(`/students/${id}`, {
    method: 'PUT',
    body: JSON.stringify(payload),
  });
}

export async function deleteStudent(id: string) {
  return apiClient<{ success: boolean; message: string }>(`/students/${id}`, {
    method: 'DELETE',
  });
}

export { AppApiError };

import {
  useQuery,
  useMutation,
  useQueryClient,
  type UseMutationOptions,
} from '@tanstack/react-query';
import {
  createStudent,
  deleteStudent,
  getStudentById,
  listStudents,
  updateStudent,
} from '@/features/students/api';
import type { ListStudentsParams, Student, StudentInput } from '@/features/students/types';
import { useToast } from '@/components/ui/toast';
import { AppApiError } from '@/lib/api-client';

export const studentsKeys = {
  all: ['students'] as const,
  list: (params: ListStudentsParams) =>
    [...studentsKeys.all, 'list', params] as const,
  detail: (id: string) => [...studentsKeys.all, 'detail', id] as const,
};

/** List query with server-side pagination, search, and filters. */
export function useStudents(params: ListStudentsParams = {}) {
  return useQuery({
    queryKey: studentsKeys.list(params),
    queryFn: () => listStudents(params),
    placeholderData: (previousData) => previousData,
  });
}

/** Single student detail query. */
export function useStudent(id?: string) {
  return useQuery({
    queryKey: studentsKeys.detail(id ?? ''),
    queryFn: () => getStudentById(id!),
    enabled: Boolean(id),
  });
}

/**
 * Create student mutation.
 * On success: invalidate student queries + toast.
 * On error: 400/409 field errors and 404/network toasts are handled by the
 * caller via mutateAsync rejection so they can be shown next to the right fields.
 */
export function useCreateStudent() {
  const queryClient = useQueryClient();
  const toast = useToast();
  return useMutation({
    mutationFn: async (payload: StudentInput) => {
      const res = await createStudent(payload);
      return res.data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: studentsKeys.all });
      toast.success('Student registered');
    },
    onError: () => {
      // Intentional no-op: the caller handles 400/409 as field errors and
      // 404/network as toasts via mutateAsync rejection.
    },
  } as UseMutationOptions<Student, unknown, StudentInput>);
}

/**
 * Update student mutation.
 * On success: invalidate student queries + toast.
 * On error: handled by the caller via mutateAsync rejection.
 */
export function useUpdateStudent() {
  const queryClient = useQueryClient();
  const toast = useToast();
  return useMutation({
    mutationFn: async ({
      id,
      payload,
    }: {
      id: string;
      payload: Partial<StudentInput>;
    }) => {
      const res = await updateStudent(id, payload);
      return res.data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: studentsKeys.all });
      toast.success('Student updated');
    },
    onError: () => {
      // Intentional no-op: the caller handles errors via mutateAsync rejection.
    },
  } as UseMutationOptions<Student, unknown, { id: string; payload: Partial<StudentInput> }>);
}

/**
 * Delete student mutation.
 * On success: invalidate student queries + toast.
 * On error: 404 and network failures are shown as toasts.
 */
export function useDeleteStudent() {
  const queryClient = useQueryClient();
  const toast = useToast();
  return useMutation({
    mutationFn: async (id: string) => {
      const res = await deleteStudent(id);
      return res.data;
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: studentsKeys.all });
      toast.success('Student deleted');
    },
    onError: (error: unknown) => {
      if (error instanceof AppApiError) {
        if (error.statusCode === 404) {
          toast.error('Not found', 'The requested student could not be found.');
          return;
        }
        toast.error('Request failed', error.message);
        return;
      }
      toast.error(
        'Network error',
        'Unable to communicate with the lunch management server.'
      );
    },
  } as UseMutationOptions<{ success: boolean; message: string }, unknown, string>);
}

'use client';

import * as React from 'react';
import Link from 'next/link';
import { LucideIcon, TrendingUp, TrendingDown, ArrowRight } from 'lucide-react';
import { cn } from '@/lib/utils/cn';

export interface SummaryCardProps {
  title: string;
  value: string | number;
  subtitle?: string;
  icon: LucideIcon;
  href?: string;
  trend?: {
    value: string | number;
    direction: 'up' | 'down' | 'neutral';
    label?: string;
  };
  variant?: 'default' | 'success' | 'warning' | 'critical' | 'navy';
  className?: string;
}

export function SummaryCard({
  title,
  value,
  subtitle,
  icon: Icon,
  href,
  trend,
  variant = 'default',
  className,
}: SummaryCardProps) {
  const iconVariants = {
    default: 'bg-slate-100 text-slate-700',
    success: 'bg-emerald-100 text-emerald-800',
    warning: 'bg-amber-100 text-amber-800',
    critical: 'bg-rose-100 text-rose-800',
    navy: 'bg-slate-900 text-white',
  };

  const borderVariants = {
    default: 'hover:border-slate-300',
    success: 'hover:border-emerald-300',
    warning: 'hover:border-amber-300',
    critical: 'hover:border-rose-300',
    navy: 'hover:border-slate-800',
  };

  const content = (
    <div
      className={cn(
        'group relative rounded-xl border border-slate-200/90 bg-white p-5 shadow-xs transition-all duration-200 hover:shadow-md cursor-pointer select-none',
        borderVariants[variant],
        className
      )}
    >
      <div className="flex items-start justify-between">
        <div className="space-y-1">
          <p className="text-xs font-semibold uppercase tracking-wider text-slate-500">
            {title}
          </p>
          <div className="flex items-baseline gap-2">
            <span className="text-2xl font-bold tracking-tight text-slate-900">
              {value}
            </span>
          </div>
        </div>
        <div
          className={cn(
            'flex h-10 w-10 shrink-0 items-center justify-center rounded-xl transition-transform group-hover:scale-105',
            iconVariants[variant]
          )}
        >
          <Icon className="h-5 w-5" />
        </div>
      </div>

      {(subtitle || trend || href) && (
        <div className="mt-3.5 flex items-center justify-between border-t border-slate-100 pt-3 text-xs text-slate-500">
          <div className="flex items-center gap-1.5">
            {trend && (
              <span
                className={cn(
                  'inline-flex items-center gap-0.5 font-medium',
                  trend.direction === 'up'
                    ? 'text-emerald-700'
                    : trend.direction === 'down'
                    ? 'text-rose-700'
                    : 'text-slate-600'
                )}
              >
                {trend.direction === 'up' && <TrendingUp className="h-3.5 w-3.5" />}
                {trend.direction === 'down' && <TrendingDown className="h-3.5 w-3.5" />}
                {trend.value}
              </span>
            )}
            {subtitle && <span>{subtitle}</span>}
          </div>

          {href && (
            <span className="inline-flex items-center gap-1 text-xs font-semibold text-emerald-800 group-hover:translate-x-0.5 transition-transform">
              View <ArrowRight className="h-3 w-3" />
            </span>
          )}
        </div>
      )}
    </div>
  );

  if (href) {
    return <Link href={href}>{content}</Link>;
  }

  return content;
}
